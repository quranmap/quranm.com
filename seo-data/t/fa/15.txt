["و (ما) آسمان‌ها و زمین و آنچه را که میان آن دو است، جز به حق نیافریدیم، و قیامت قطعاً خواهد آمد، پس (ای پیامبر) به طرز شایسته‌ای گذشت کن (و از آن‌ها صرف نظر فرما)","و ما آسمان‌ها و زمین و آنچه را که میان آنهاست جز به‌حق نیافریده‌ایم؛ و قیامت قطعاً آمدنی است؛ پس [ای پیامبر، از کافران‌] به نیکویی گذشت کن.‌"]
["بی‌گمان پروردگار تو، همان آفرینندۀ داناست.","بی‌تردید، پروردگارت همان آفریدگارِ داناست. ‌"]
["و به راستی (ما) به تو سبع المثانی (= سورۀ فاتحه) و قرآن عظیم دادیم.","و به راستی، ما به تو «سبعُ المثانی» [= سورۀ فاتحه] و قرآن بزرگ را دادیم."]
["(پس) هرگز چشم خود را به (سوی) آنچه (از نعمت‌ها و متاع دنیوی) که گروه‌هایی از آنان را از آن بهرمند ساخته‌ایم، نیندوز، و بر آن‌ها غمگین نباش، و بال (شفقت) خود را برای مؤمنان فرود آر (و نسبت به آن‌ها فروتن باش).","[بنابراین،] هرگز به آنچه [از نعمت‌های دنیوی] که گروه‌هایی از آنان [= کافران] را بهره‌مند کرده‌ایم چشم ندوز و به خاطر [ناسپاسی و شرکِ] آنان اندوهگین نباش و با مؤمنان فروتنی کن."]
["و بگو: «همانا من هشدار دهندۀ آشکارم».","و بگو: «به راستی که من همان بیم‌دهندۀ آشکارم»."]
["(و آن‌ها را از عذابی بترسان) همان‌گونه که بر تقسیم‌کنندگان (آیات الهی) نازل کردیم.","[و آنان را از عذاب بترسان] همان گونه که [آن را] بر تقسیم‌کنندگان [آیات الهی] فرستادیم؛"]
["(همان) کسانی‌که قرآن را تقسیم (و بخش، بخش) کردند.","همان کسانی كه قرآن را بخش بخش کردند [و آن را شعر، سحر، کذب، و اساطیر ‌نامیدند]."]
["پس سوگند به پروردگارت، قطعاً (در قیامت) از همه آن‌ها سوال خواهیم کرد.","سوگند به پروردگارت، قطعاً [در قیامت] از همۀ آنان سؤال خواهیم کرد،"]
["از آنچه می‌کردند.","از آنچه می‌کردند."]
["پس (ای پیامبر!) به آنچه مأمور شده‌ای آشکارا ابلاغ کن، و از مشرکان روی بگردان.","پس [ای پیامبر،] آنچه [را که از سوی الله] دستور یافته‌ای، آشکار کن و از مشرکان روی بگردان‌."]
["بی‌گمان ما (شر) مسخره کنندگان را از تو دفع خواهیم کرد.","بی‌تردید، ما [شرّ] مسخره‌کنندگان را از تو دور می‌کنیم."]
["کسانی‌که معبود دیگری با الله قرار می‌دهند، پس بزودی خواهند دانست.","همان کسانی که [برای خود،] با الله، معبود دیگری قرار می‌دهند؛ پس به زودی [نتیجۀ شرکِ خویش را] خواهند دانست."]
["و به راستی ما می‌دانیم که سینه تو (ای پیامبر) از آنچه آن‌ها می‌گویند تنگ می‌شود (و تو را آزرده و نا راحت می‌کند).","و [ای پیامبر،] یقیناً ما می‌دانیم که تو از آنچه آنان [= کافران] می‌گویند آزرده‌خاطر می‌گردی."]
["پس به ستایش پروردگارت تسبیح گوی و از سجده‌کنندگان باش.","پس به ستایش پروردگارت تسبیح بگو و از سجده‌کنندگان باش."]
["و پروردگارت را عبادت کن تا هنگامی‌که یقین (= مرگ) به سراغت آید.","و تا هنگامی‌ که مرگت فرامی‌رسد پروردگارت را عبادت کن."]

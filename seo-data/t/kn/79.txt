["ಆಗ ಅಲ್ಲಾಹು ಅವನನ್ನು ಪರಲೋಕ ಮತ್ತು ಇಹಲೋಕದ ಶಿಕ್ಷೆಗಾಗಿ ಹಿಡಿದನು.","ಕೊನೆಗೆ ಅಲ್ಲಾಹನು ಆತನನ್ನು ಪಾರತ್ರಿಕ ಮತ್ತು ಐಹಿಕ ಶಿಕ್ಷೆಗಳಿಗೆ ಗುರಿಪಡಿಸಿದನು."]
["ನಿಶ್ಚಯವಾಗಿಯೂ ಭಯಪಡುವವರಿಗೆ ಅದರಲ್ಲಿ ನೀತಿಪಾಠವಿದೆ.","ಈ ಹೇಳಿದುದರಲ್ಲಿ ಅಲ್ಲಾಹನನ್ನು ಭಯ ಪಡುವವರಿಗೆ ಖಂಡಿತ ಪಾಠವಿದೆ."]
["ನಿಮ್ಮನ್ನು ಸೃಷ್ಟಿಸುವುದು ಕಷ್ಟವೋ ಅಥವಾ ಆಕಾಶವನ್ನೋ? ಅವನು ಆಕಾಶವನ್ನು ನಿರ್ಮಿಸಿದನು.","ನಿಮ್ಮನ್ನು ಸೃಷ್ಟಿಸುವುದೋ ಅಥವಾ ಆಕಾಶವನ್ನು ಸೃಷ್ಟಿಸುವುದೋ ಹೆಚ್ಚು ಪ್ರಯಾಸಕರ? ಅವನು ಆಕಾಶವನ್ನು ಸೃಷ್ಟಿಸಿದನು."]
["ಅದರ ಛಾವಣಿಯನ್ನು ಎತ್ತರಿಸಿದನು ಮತ್ತು ಅದನ್ನು ಸರಿಪಡಿಸಿದನು.","ಅದರ ಛಾವಣಿಯನ್ನು ಎತ್ತರಿಸಿದನು ಹಾಗೂ ಅದನ್ನು ಸಮರ್ಪಕಗೊಳಿಸಿದನು."]
["ಅದರ ರಾತ್ರಿಯನ್ನು ಕತ್ತಲೆಯಾಗಿ ಮಾಡಿದನು ಮತ್ತು ಅದರ ಹಗಲನ್ನು ಪ್ರತ್ಯಕ್ಷಗೊಳಿಸಿದನು.","ಅವನು ಅದರ ರಾತ್ರಿಗೆ ಕತ್ತಲು ಕವಿಸಿದನು ಹಾಗೂ ಅದರ ಬೆಳಕನ್ನು ಪ್ರಕಟಗೊಳಿಸಿದನು."]
["ಅದರ ನಂತರ ಅವನು ಭೂಮಿಯನ್ನು ವಿಸ್ತರಿಸಿದನು.","ಅದರ ನಂತರ ಭೂಮಿಯನ್ನು ಹಾಸಿದನು."]
["ಅದರಿಂದ ನೀರನ್ನು ಮತ್ತು ಹುಲ್ಲುಗಾವಲುಗಳನ್ನು ಹೊರತೆಗೆದನು.","ಅದರಿಂದ ಅದರ ನೀರನ್ನೂ ಮೇವನ್ನೂ ಹೊರಡಿಸಿದನು."]
["ಪರ್ವತಗಳನ್ನು ದೃಢವಾಗಿ ನಿಲ್ಲಿಸಿದನು.","ಅದರಲ್ಲಿ ಪರ್ವತಗಳನ್ನು ನಾಟಿದನು."]
["ಇವೆಲ್ಲವೂ ನಿಮ್ಮ ಮತ್ತು ನಿಮ್ಮ ಜಾನುವಾರುಗಳ ಉಪಯೋಗಕ್ಕಾಗಿ.","ನಿಮ್ಮ ಹಾಗೂ ನಿಮ್ಮ ಜಾನುವಾರುಗಳ ಪ್ರಯೋಜನಕ್ಕಾಗಿ."]
["ಆ ಮಹಾ ವಿಪತ್ತು (ಪುನರುತ್ಥಾನ ದಿನ) ಬರುವ ಸಂದರ್ಭ.","ಮುಂದೆ ಅತೀ ಭಯಂಕರವಾದ ವಿಪತ್ತು ಬಂದಾಗ,"]
["ಅಂದು ಮನುಷ್ಯನು ತಾನು ಮಾಡಿಟ್ಟ ಕರ್ಮಗಳನ್ನು ಸ್ಮರಿಸುವನು.","ಮನುಷ್ಯನು ತಾನು ಮಾಡಿದ್ದನ್ನು ಸ್ಮರಿಸುವ ದಿನವದು!"]
["ನೋಡುಗರೆಲ್ಲರ ಮುಂದೆ ನರಕವನ್ನು ಪ್ರದರ್ಶಿಸಲಾಗುವುದು.","ಜ್ವಲಿಸುವ ನರಕವು ನೋಡುಗರಿಗೆ ಪ್ರದರ್ಶಿಸಲ್ಪಡುವ ದಿನ."]
["ಆದ್ದರಿಂದ ಯಾರು ಎಲ್ಲೆ ಮೀರಿದವನೋ,","ಯಾವನು ದುರ್ಮಾರ್ಗಿಯಾಗಿರುವನೋ,"]
["ಮತ್ತು ಇಹಲೋಕಕ್ಕೆ ಪ್ರಾಶಸ್ತ್ಯ ನೀಡಿದವನೋ,","ಹಾಗೂ ಕೇವಲ ಭೌತಿಕ ಜೀವನವನ್ನು ಆಯ್ದು ಕೊಂಡಿರುವನೋ,"]
["ನಿಶ್ಚಯವಾಗಿಯೂ ನರಕವೇ ಅವನ ವಾಸಸ್ಥಳ.","ಖಂಡಿತವಾಗಿಯೂ ನರಕವೇ ಅವನ ಅಂತಿಮ ನೆಲೆಯಾಗಿರುವುದು."]
["ಯಾರು ತನ್ನ ಪರಿಪಾಲಕನ (ಅಲ್ಲಾಹನ) ಮುಂದೆ ನಿಲ್ಲುವುದನ್ನು ಭಯಪಟ್ಟನೋ ಮತ್ತು ತನ್ನ ದೇಹವನ್ನು ಸ್ವೇಚ್ಛೆಗಳಿಂದ ದೂರವಿಟ್ಟವನೋ,","ಯಾವನು ತನ್ನ ಪ್ರಭುವಿನ ಸನ್ನಿಧಾನವನ್ನು ಭಯಪಟ್ಟಿರುವನೋ ಹಾಗೂ ದೇಹವನ್ನು ಕಾಮನೆಗಳಿಂದ ದೂರವಿರಿಸಿಕೊಳ್ಳುವನೋ,"]
["ನಿಶ್ಚಯವಾಗಿಯೂ ಸ್ವರ್ಗವೇ ಅವನ ವಾಸಸ್ಥಳ.","ಖಂಡಿತ, ಸ್ವರ್ಗವೇ ಅವನ ಅಂತಿಮ ನೆಲೆಯಾಗಿರುವುದು."]
["ಅಂತ್ಯಸಮಯವು—ಅದು ಯಾವಾಗ ಸಂಭವಿಸುತ್ತದೆಯೆಂದು ಅವರು ನಿಮ್ಮಲ್ಲಿ ಕೇಳುತ್ತಾರೆ.","ಅವರು ಅಂತ್ಯದಿನ ಬರುವುದು ಯಾವಾಗ ಎಂದು ತಮ್ಮಲ್ಲಿ ಪ್ರಶ್ನಿಸುತ್ತಾರೆ."]
["ನಿಮಗೆ ಅದರ ಬಗ್ಗೆ ಏನು ಜ್ಞಾನವಿದೆ?","ಅದನ್ನು ಅವರಿಗೆ ತಿಳಿಸಿಕೊಡುವ ಅರಿವು ತಮಗೆಲ್ಲಿದೆ?"]
["ಅದರ ಆತ್ಯಂತಿಕ ಜ್ಞಾನವಿರುವುದು ನಿಮ್ಮ ಪರಿಪಾಲಕನಿಗೆ (ಅಲ್ಲಾಹನಿಗೆ) ಮಾತ್ರ.","ಅದರ ಅರಿವಿನ ಪಾರಮ್ಯವು ನಿಮ್ಮ ಪ್ರಭುವಿನ ಕಡೆಗಿದೆ."]
["ನೀವಂತೂ ಅದನ್ನು ಭಯಪಡುವವರಿಗೆ ಮುನ್ನೆಚ್ಚರಿಕೆ ನೀಡುವವರಾಗಿದ್ದೀರಿ.","ಅಂತ್ಯದಿನವನ್ನು ಭಯಪಡುವವರಿಗೆ ತಾವು ಎಚ್ಚರಿಕೆಯನ್ನು ನೀಡುವವರು ಮಾತ್ರವಾಗಿರುವಿರಿ."]
["ಅವರು ಅದನ್ನು ನೋಡುವ ದಿನದಂದು, ದಿನದ ಕೊನೆಯ ಭಾಗ ಅಥವಾ ದಿನದ ಮೊದಲ ಭಾಗ ಮಾತ್ರ ಅವರು (ಭೂಮಿಯಲ್ಲಿ) ವಾಸವಾಗಿದ್ದರು ಎಂದು ಅವರಿಗೆ ಭಾಸವಾಗುವುದು.","ಅಂತ್ಯದಿನವನ್ನು ಅವರು ಕಣ್ಣಾರೆ ಕಾಣುವ ದಿನ ನಾವು ಭೂಮಿಯಲ್ಲಿ ತಂಗಿದ್ದು ಕೇವಲ ಸಂಜೆ ಅಥವಾ ಮುಂಜಾನೆ ಮಾತ್ರವಾಗಿದೆಯೆಂದು ಅವರು ಭಾವಿಸುವರು."]
